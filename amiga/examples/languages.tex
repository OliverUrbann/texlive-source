% lualatex languages.tex -- needs the font add-on TeX-Fonts-Europe.lha.
% Five languages, one font (CMU = Computer Modern Unicode). Each language
% gets its own hyphenation. With babel, every language loads the font
% anew (other script, other features), so a light font matters: with
% FreeSerif this document needs too much memory for the Amiga.
\documentclass{article}
\usepackage{babel}
\babelprovide[import, main]{german}
\babelprovide[import]{russian}
\babelprovide[import]{greek}
\babelprovide[import]{polish}
\babelprovide[import]{french}
\babelfont{rm}{CMU Serif}
\begin{document}
\section{Mehrere Sprachen}
Deutsch: Donaudampfschifffahrtsgesellschaftskapitän.

\foreignlanguage{russian}{Русский: Достопримечательности государственного университета.}

\foreignlanguage{greek}{Ελληνικά: Καλημέρα κόσμε, ηλεκτρονικός υπολογιστής.}

\foreignlanguage{polish}{Polski: Zażółć gęślą jaźń, konstantynopolitańczykowianeczka.}

\foreignlanguage{french}{Français : anticonstitutionnellement, cœur, élève.}

\showhyphens{\selectlanguage{russian}Достопримечательности}
\showhyphens{\selectlanguage{polish}konstantynopolitańczykowianeczka}
\showhyphens{\selectlanguage{greek}ηλεκτρονικός}
\end{document}
